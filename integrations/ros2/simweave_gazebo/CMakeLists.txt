cmake_minimum_required(VERSION 3.16)
project(simweave_gazebo)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(simulation_interfaces REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "srv/ConfigureCart.srv" "srv/SetCartForce.srv" "srv/GetCartState.srv"
  DEPENDENCIES simulation_interfaces)
add_executable(cart_server src/cart_server.cpp)
target_compile_features(cart_server PRIVATE cxx_std_17)
target_compile_options(cart_server PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(cart_server rclcpp simulation_interfaces)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} "rosidl_typesupport_cpp")
target_link_libraries(cart_server "${typesupport}" gz-sim8::gz-sim8)
install(TARGETS cart_server DESTINATION lib/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
